{{- /* Kõik lingid tekstis (mistahes lehel) - kui link viitab YouTube'i videole, asendatakse nähtav tekstilink thumbnailiga (koos tegeliku pealkirjaga YouTube'ist selle all). Muud lingid renderdatakse tavapäraselt, ilma muudatusteta. */ -}} {{- $dest := .Destination -}} {{- $yt := partial "youtube-item.html" (dict "url" $dest) -}} {{- if $yt.id -}} {{- $title := "" -}} {{- $watchURL := printf "https://www.youtube.com/watch?v=%s" $yt.id -}} {{- $oembedURL := printf "https://www.youtube.com/oembed?url=%s&format=json" (urlquery $watchURL) -}} {{- $res := try (resources.GetRemote $oembedURL) -}} {{- with $res.Err -}} {{- warnf "YouTube oEmbed ebaõnnestus (%s): %s" $watchURL . -}} {{- else with $res.Value -}} {{- $data := . | transform.Unmarshal -}} {{- $title = $data.title | default "" -}} {{- end -}} {{- with $title }}{{ . }}{{ end -}} {{- else -}} {{ .Text }} {{- end -}}
