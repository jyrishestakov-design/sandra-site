{{- /* Kõik lingid tekstis (mistahes lehel) - kui link viitab YouTube'i videole, asendatakse nähtav tekstilink thumbnailiga selle all oleva pealkirjaga. Kui toimetaja on lingile ise teksti kirjutanud (nt "Kultuuritehas. ... 16.03.2012" - kuupäevaga), kasutatakse seda teksti (see on igal ajal ise muudetav lingi teksti muutes). Kui link on lihtsalt puhas URL (toimetaja pole oma teksti kirjutanud), tõmmatakse pealkiri automaatselt YouTube'ist. Muud (mitte-YouTube) lingid renderdatakse tavapäraselt. */ -}} {{- $dest := .Destination -}} {{- $yt := partial "youtube-item.html" (dict "url" $dest) -}} {{- if $yt.id -}} {{- $plain := strings.TrimSuffix "/" (strings.TrimSpace .PlainText) -}} {{- $destTrimmed := strings.TrimSuffix "/" $dest -}} {{- $caption := "" -}} {{- if and $plain (ne $plain $destTrimmed) -}} {{- $caption = .Text -}} {{- else -}} {{- $watchURL := printf "https://www.youtube.com/watch?v=%s" $yt.id -}} {{- $oembedURL := printf "https://www.youtube.com/oembed?url=%s&format=json" (urlquery $watchURL) -}} {{- $res := try (resources.GetRemote $oembedURL) -}} {{- with $res.Err -}} {{- warnf "YouTube oEmbed ebaõnnestus (%s): %s" $watchURL . -}} {{- else with $res.Value -}} {{- $data := . | transform.Unmarshal -}} {{- $caption = $data.title | default "" -}} {{- end -}} {{- end -}} {{- with $caption }}{{ . }}{{ end -}} {{- else -}} {{ .Text }} {{- end -}}
